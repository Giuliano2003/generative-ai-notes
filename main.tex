% Illustrazione simbolica:
% dal prompt alla generazione attraverso una rete neurale
\begin{tikzpicture}[
    x=1cm,
    y=1cm,
    >=Stealth,
    neuron/.style={
        circle,
        draw=coverblue,
        line width=0.8pt,
        fill=white,
        minimum size=5.5mm,
        inner sep=0pt
    },
    token/.style={
        rounded corners=3pt,
        draw=coverblue!20,
        fill=coverlight,
        minimum width=1.7cm,
        minimum height=0.65cm,
        font=\small,
        text=coverblue
    }
]

    % Fondo dell'illustrazione
    \fill[
        coverlight!65,
        rounded corners=18pt
    ] (-0.7,-2.05) rectangle (6.7,2.05);

    % Collegamenti fra i livelli della rete
    \foreach \ya in {-1.1,0,1.1} {
        \foreach \yb in {-1.5,-0.5,0.5,1.5} {
            \draw[
                coverblue!22,
                line width=0.6pt
            ] (0.6,\ya) -- (3,\yb);

            \draw[
                coverblue!22,
                line width=0.6pt
            ] (3,\yb) -- (5.4,\ya);
        }
    }

    % Alcuni percorsi in evidenza
    \draw[
        coveraccent,
        line width=1.1pt,
        opacity=0.8
    ] (0.6,0) -- (3,0.5) -- (5.4,1.1);

    \draw[
        coveraccent,
        line width=1.1pt,
        opacity=0.8
    ] (0.6,-1.1) -- (3,-0.5) -- (5.4,0);

    % Neuroni dei livelli esterni
    \foreach \y in {-1.1,0,1.1} {
        \node[neuron] at (0.6,\y) {};
        \node[neuron] at (5.4,\y) {};
    }

    % Neuroni del livello centrale
    \foreach \y in {-1.5,-0.5,0.5,1.5} {
        \node[
            neuron,
            fill=coverlight
        ] at (3,\y) {};
    }

    % Nodi attivi
    \node[
        neuron,
        draw=coveraccent,
        fill=coveraccent!18
    ] at (3,0.5) {};

    \node[
        neuron,
        draw=coveraccent,
        fill=coveraccent!18
    ] at (5.4,1.1) {};

    % Prompt
    \node[token] (prompt) at (-2.15,0)
        {Prompt};

    \draw[
        ->,
        coverblue!60,
        thick
    ] (prompt.east) -- (0.1,0);

    % Output generato
    \node[
        token,
        draw=coveraccent!35,
        fill=coveraccent!10,
        text=coveraccent!80!black,
        align=center
    ] (output) at (8.15,0)
        {Testo\\generato};

    \draw[
        ->,
        coveraccent,
        thick
    ] (5.9,0) -- (output.west);

    % Didascalia interna
    \node[
        font=\scriptsize,
        text=coverblue!65
    ] at (3,-2.5)
        {Rappresentazioni apprese e generazione};

\end{tikzpicture}
